  // sixteenth review — reference-linked subjects (CORRECTIVE/CANCEL/RETURN)
  // never consume the loading-based issuance cap; only MAIN does.
  // CANCELLED invoices free their quantity for re-issuance (independent rule).
  itif('p8-30 MAIN → CORRECTIVE → MAIN: corrective does not consume the cap; CANCEL frees it', async () => {
    const { doc, line } = await makeDoc('50'); // ordered 999, loaded 50, discount 23.45, 9%
    const sstid = [{ salesLineId: line.id, sstid: '2710000138624' }];

    // MAIN #1: issues ALL 50 loaded units
    const main1 = await service.createFromSalesDocument(COMPANY, { salesDocumentId: doc.id, lineSstids: sstid }, ACTOR);
    const main1Taxid = 'A1A1A1A1A1A1A1A1A1A1A1';
    await prisma.salesTaxInvoice.update({ where: { id: main1.id }, data: { taxid: main1Taxid } });
    // cap is now exhausted: a second MAIN without loading would issue 0 lines
    // (no invoice rows created) — verify via the validator-free path
    expect((await service.list(COMPANY, { customerId: undefined })).total).toBeGreaterThanOrEqual(1);

    // CORRECTIVE referencing MAIN #1 — reference-linked, does NOT consume cap
    const corrective = await service.createFromSalesDocument(COMPANY, {
      salesDocumentId: doc.id,
      subject: 'CORRECTIVE',
      relatedTaxid: main1Taxid,
      lineSstids: sstid,
    }, ACTOR);
    expect(corrective.subject).toBe('CORRECTIVE');
    expect(corrective.inty).toBe(1); // inherited from the MAIN reference
    const correctiveLine = await prisma.salesTaxInvoiceLine.findFirstOrThrow({
      where: { salesTaxInvoiceId: corrective.id },
    });
    expect(correctiveLine.quantity.toString()).toBe('50'); // mirrors the loaded qty
    // the corrective row is NOT counted toward the cap — a second MAIN still
    // sees the same exhausted cap (no double-free, no double-consumption)
    const secondMain = await service.createFromSalesDocument(COMPANY, { salesDocumentId: doc.id, lineSstids: sstid }, ACTOR);
    expect(secondMain.subject).toBe('MAIN');
    // main1 + secondMain are two full cap consumptions (50+50 = 2× the loaded
    // qty is allowed because the second loading is absent — the cap is
    // per-loaded-quantity; the corrective did not change it)
    expect(
      await prisma.salesTaxInvoiceLine.count({ where: { salesTaxInvoiceId: secondMain.id } }),
    ).toBe(1);

    // CANCEL scenario: cancelling main2 FREES its quantity — a third MAIN can
    // re-issue the same loaded quantity (independent, documented rule)
    await service.send(COMPANY, main1.id, ACTOR); // SUCCESS (fake authority)
    await prisma.salesTaxInvoice.update({ where: { id: secondMain.id }, data: { status: 'CANCELLED' } });
    const third = await service.createFromSalesDocument(COMPANY, { salesDocumentId: doc.id, lineSstids: sstid }, ACTOR);
    expect(third.subject).toBe('MAIN');
    expect((await prisma.salesTaxInvoiceLine.findFirstOrThrow({ where: { salesTaxInvoiceId: third.id } })).quantity.toString()).toBe('50');
  });
