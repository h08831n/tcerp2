  itif(
    'p8-33a referenced-original update failure rolls back the WHOLE cancellation transaction',
    async () => {
      const { doc, line } = await makeDoc('50');
      const sstid = [{ salesLineId: line.id, sstid: '2710000138624' }];
      const main = await service.createFromSalesDocument(COMPANY, { salesDocumentId: doc.id, lineSstids: sstid }, ACTOR);
      const mainTaxid = 'D2D2D2D2D2D2D2D2D2D2D2';
      await prisma.salesTaxInvoice.update({ where: { id: main.id }, data: { taxid: mainTaxid } });
      await service.send(COMPANY, main.id, ACTOR);
      fakeAdapter.inquiryQueue.push('SUCCESS');
      await service.inquiry(COMPANY, main.id);

      const cancelInvoice = await service.createFromSalesDocument(COMPANY, {
        salesDocumentId: doc.id,
        subject: 'CANCEL',
        relatedTaxid: mainTaxid,
        lineSstids: sstid,
      }, ACTOR);
      await service.send(COMPANY, cancelInvoice.id, ACTOR);

      // inject a failure into the REFERENCED-ORIGINAL update (updateMany)
      // INSIDE the inquiry transaction — the submission+invoice SUCCESS
      // writes and the original cancellation must roll back TOGETHER
      const realTransaction = prisma.$transaction.bind(prisma);
      let injectionHits = 0;
      const txSpy = jest
        .spyOn(prisma, '$transaction')
        .mockImplementationOnce((async (arg: unknown) => {
          return realTransaction(async (tx: Prisma.TransactionClient) => {
            const failingTx = new Proxy(tx, {
              get(target, prop) {
                if (prop === 'salesTaxInvoice') {
                  return new Proxy(target.salesTaxInvoice, {
                    get(t, p) {
                      if (p === 'updateMany') {
                        return (...args: unknown[]) => {
                          injectionHits += 1;
                          return Promise.reject(new Error('INJECTED: referenced-original update failure'));
                        };
                      }
                      const v = Reflect.get(t, p, t);
                      return typeof v === 'function' ? (v as (...a: unknown[]) => unknown).bind(t) : v;
                    },
                  });
                }
                return Reflect.get(target, prop);
              },
            });
            return (arg as (tx: Prisma.TransactionClient) => unknown)(failingTx);
          });
        }) as never);
      fakeAdapter.inquiryQueue.push('SUCCESS');
      await expect(service.inquiry(COMPANY, cancelInvoice.id)).rejects.toThrow('INJECTED');
      txSpy.mockRestore();
      expect(injectionHits).toBe(1); // the hook WAS reached, on the original's updateMany

      // FULL rollback: cancellation submission stays PENDING, the cancel
      // invoice stays SUBMITTED, the referenced original is untouched
      const submission = await prisma.salesTaxInvoiceSubmission.findFirstOrThrow({
        where: { invoiceId: cancelInvoice.id },
        orderBy: { attempt: 'desc' },
      });
      expect(submission.status).toBe('PENDING');
      const cancelRow = await prisma.salesTaxInvoice.findUniqueOrThrow({ where: { id: cancelInvoice.id } });
      expect(cancelRow.status).toBe('SUBMITTED');
      const originalRow = await prisma.salesTaxInvoice.findUniqueOrThrow({ where: { id: main.id } });
      expect(originalRow.status).not.toBe('CANCELLED'); // remains consumed until confirmed

      // retry WITHOUT injection: everything applies together
      fakeAdapter.inquiryQueue.push('SUCCESS');
      expect((await service.inquiry(COMPANY, cancelInvoice.id)).status).toBe('SUCCESS');
      const cancelledNow = await prisma.salesTaxInvoice.findUniqueOrThrow({ where: { id: main.id } });
      expect(cancelledNow.status).toBe('CANCELLED');
      const confirmedCancel = await prisma.salesTaxInvoice.findUniqueOrThrow({ where: { id: cancelInvoice.id } });
      expect(confirmedCancel.status).toBe('SUCCESS');
      expect(
        await prisma.salesTaxInvoiceSubmission.count({ where: { invoiceId: cancelInvoice.id, status: 'SUCCESS' } }),
      ).toBe(1);
    },
  );

  itif(
    'p8-33b recovery reconciles a torn SUCCESS cancellation idempotently — no enqueue, referenced original cancelled',
    async () => {
      const { doc, line } = await makeDoc('50');
      const sstid = [{ salesLineId: line.id, sstid: '2710000138624' }];
      const main = await service.createFromSalesDocument(COMPANY, { salesDocumentId: doc.id, lineSstids: sstid }, ACTOR);
      const mainTaxid = 'E2E2E2E2E2E2E2E2E2E2E2';
      await prisma.salesTaxInvoice.update({ where: { id: main.id }, data: { taxid: mainTaxid } });
      await service.send(COMPANY, main.id, ACTOR);
      fakeAdapter.inquiryQueue.push('SUCCESS');
      await service.inquiry(COMPANY, main.id);

      const cancelInvoice = await service.createFromSalesDocument(COMPANY, {
        salesDocumentId: doc.id,
        subject: 'CANCEL',
        relatedTaxid: mainTaxid,
        lineSstids: sstid,
      }, ACTOR);
      await service.send(COMPANY, cancelInvoice.id, ACTOR);
      // torn state: the authority accepted (submission SUCCESS persisted) but
      // the local repair never ran — invoice stale SENDING, original uncancelled
      await prisma.salesTaxInvoiceSubmission.updateMany({
        where: { invoiceId: cancelInvoice.id },
        data: { status: 'SUCCESS', confirmationReferenceId: 'conf-cancel' },
      });
      await prisma.salesTaxInvoice.update({ where: { id: cancelInvoice.id }, data: { status: 'SENDING' } });
      const enqueuedBefore = fakeAdapter.enqueued.length;

      // recovery path (SENDING → recoverPending, no PENDING submissions)
      const recovered = await service.send(COMPANY, cancelInvoice.id, ACTOR);
      expect(recovered.status).toBe('SUCCESS');
      expect(recovered.recoveredLateSuccess).toBe(true);
      expect(fakeAdapter.enqueued.length).toBe(enqueuedBefore); // NO re-submission
      const repaired = await prisma.salesTaxInvoice.findUniqueOrThrow({ where: { id: cancelInvoice.id } });
      expect(repaired.status).toBe('SUCCESS');
      const freedOriginal = await prisma.salesTaxInvoice.findUniqueOrThrow({ where: { id: main.id } });
      expect(freedOriginal.status).toBe('CANCELLED'); // reconciled by the recovery
      // moadianStatus keeps its documented meaning on the ORIGINAL
      expect(freedOriginal.moadianStatus).toBe('SUCCESS');

      // idempotent: a repeated recovery repairs nothing further and is safe
      await prisma.salesTaxInvoice.update({ where: { id: cancelInvoice.id }, data: { status: 'SENDING' } });
      const again = await service.send(COMPANY, cancelInvoice.id, ACTOR);
      expect(again.status).toBe('SUCCESS');
      expect(fakeAdapter.enqueued.length).toBe(enqueuedBefore);
      expect((await prisma.salesTaxInvoice.findUniqueOrThrow({ where: { id: main.id } })).status).toBe('CANCELLED');
    },
